#
# Helper programs for the shell based test suite. Matches noinst_PROGRAMS in
# tests/Makefile.am: they are built but never installed.
#
# The test scripts expect the autotools in-tree layout (./prog_foo,
# ../utils/iscsi-inq, ../test-tool/iscsi-test-cu). CMake already puts each
# target in the build directory that matches its source directory, so we only
# have to copy the scripts across for the layout to line up.
#
set(ISCSI_TEST_PROGS
    prog_reconnect
    prog_reconnect_timeout
    prog_noop_reply
    prog_readwrite_iov
    prog_timeout
    prog_read_all_pdus
    prog_header_digest)
# Uses the multithreading service thread, and POSIX signals.
if(HAVE_PTHREAD)
    list(APPEND ISCSI_TEST_PROGS prog_mt_reconnect)
endif()

foreach(_prog IN LISTS ISCSI_TEST_PROGS)
    add_executable(${_prog} ${_prog}.c)
    libiscsi_apply_build_settings(${_prog})
    target_link_libraries(${_prog} PRIVATE iscsi ${LIBISCSI_PLATFORM_LIBS})
endforeach()

#
# "cmake --build <dir> --target check" is the equivalent of "make test" in
# tests/. The scripts drive a real target, so they need tgtd and tgtadm on
# the PATH.
#
file(GLOB _test_scripts "${CMAKE_CURRENT_SOURCE_DIR}/test_*.sh")
list(APPEND _test_scripts "${CMAKE_CURRENT_SOURCE_DIR}/functions.sh")

set(_staged_scripts "")
foreach(_script IN LISTS _test_scripts)
    get_filename_component(_name "${_script}" NAME)
    set(_staged "${CMAKE_CURRENT_BINARY_DIR}/${_name}")
    add_custom_command(OUTPUT "${_staged}"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${_script}" "${_staged}"
        DEPENDS "${_script}"
        COMMENT "Staging ${_name}"
        VERBATIM)
    list(APPEND _staged_scripts "${_staged}")
endforeach()

add_custom_target(stage-test-scripts DEPENDS ${_staged_scripts})

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/run-tests.sh"
"#!/bin/sh
# Generated by CMake. Runs the same tests as \"make test\" does in tests/.
for TEST in test_*.sh; do
    echo \"Running $TEST\"
    echo \"--------------\"
    sh \"$TEST\" || exit 1
    echo \"--------------\"
    echo
done
")

set(_check_depends stage-test-scripts ${ISCSI_TEST_PROGS} iscsi-inq iscsi-ls)
if(TARGET iscsi-test-cu)
    list(APPEND _check_depends iscsi-test-cu)
endif()

add_custom_target(check
    COMMAND sh run-tests.sh
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    DEPENDS ${_check_depends}
    COMMENT "Running the libiscsi test suite (requires tgtd and tgtadm)"
    VERBATIM)
